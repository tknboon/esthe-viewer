rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /sharedState/{docId} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    match /regions/{regionId}/sharedState/{docId} {
      allow read: if true;
      allow write: if request.auth != null
        && regionId in ["aichi"];
    }
  }
}
